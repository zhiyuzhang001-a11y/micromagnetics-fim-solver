cmake_minimum_required(VERSION 3.18)
project(boris_fim_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3)
option(FIM_LOCAL_TWO_OWNER "Use the adopted two-owner CPU local-field backend" ON)
if(FIM_LOCAL_TWO_OWNER)
  find_package(Threads REQUIRED)
endif()
foreach(target dynamic static inspect)
  add_executable(fim_${target} src/${target}.cpp)
  target_compile_options(fim_${target} PRIVATE -O2)
  target_link_libraries(fim_${target} PRIVATE PkgConfig::FFTW)
  if(FIM_LOCAL_TWO_OWNER)
    target_compile_definitions(fim_${target} PRIVATE LF_TWO_OWNER)
    target_link_libraries(fim_${target} PRIVATE Threads::Threads)
  endif()
endforeach()
option(BUILD_VALIDATION_TARGETS "Build separate restricted validation CLIs" OFF)
if(BUILD_VALIDATION_TARGETS)
  add_executable(label_prefix validation/label/core_label.cpp)
  target_link_libraries(label_prefix PRIVATE PkgConfig::FFTW)
  pkg_check_modules(MPFR REQUIRED IMPORTED_TARGET mpfr)
  pkg_check_modules(GMP REQUIRED IMPORTED_TARGET gmp)
  add_executable(d0_reference validation/d0/reference_point.cpp)
  target_link_libraries(d0_reference PRIVATE PkgConfig::MPFR PkgConfig::GMP)
  if(APPLE)
    add_executable(fm_d1ns validation/fm/fm_d1ns.cpp)
    target_link_libraries(fm_d1ns PRIVATE "-framework Accelerate")
  endif()
endif()
